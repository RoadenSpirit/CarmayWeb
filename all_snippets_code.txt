/**
 * Carmay Stealth Sale Tagger - BACKEND
 * Objetivo: Sincronizar productos en oferta con una etiqueta profesional para el filtro.
 */

(function() {

    // 1. FUNCIÓN DE SINCRONIZACIÓN AL GUARDAR O ACTUALIZAR
    add_action('woocommerce_update_product', function($product_id) {
        $product = wc_get_product($product_id);
        if (!$product) return;

        // DEFINICIÓN DE LA ETIQUETA
        $tag_name = 'Ofertas Especiales'; // Este es el nombre que aparecerá en el filtro
        $tag_slug = 'oferta-carmay';    // Slug técnico

        // Verificamos si la etiqueta existe, si no, la creamos
        if (!term_exists($tag_name, 'product_tag')) {
            wp_insert_term($tag_name, 'product_tag', array('slug' => $tag_slug));
        }

        // LÓGICA DE ASIGNACIÓN
        if ($product->is_on_sale()) {
            // Asignamos la etiqueta (true para no borrar las actuales)
            wp_set_object_terms($product_id, $tag_slug, 'product_tag', true);
        } else {
            // Si ya no está en oferta, removemos esta etiqueta específica
            wp_remove_object_terms($product_id, $tag_slug, 'product_tag');
        }
    }, 10, 1);

    // 2. CSS STEALTH (Ocultar solo en la ficha de producto y metas)
    add_action('wp_head', function() {
        ?>
        <style>
            /* Ocultamos la etiqueta en cualquier lugar del tema donde Blocksy la muestre */
            a[href*="oferta-carmay"], 
            .tag-link-oferta-carmay,
            .term-oferta-carmay {
                display: none !important;
            }
        </style>
        <?php
    });

})();

=== SNIPPET: Productos Trending Tag Server (ID: 2884) ===

/**
 * Carmay Trending Engine - BACKEND
 * Responsabilidad: Analizar ventas y stock para asignar etiquetas.
 */

(function() {
    // 1. Registro del evento para ejecución en segundo plano
    if (!wp_next_scheduled('carmay_cron_trending_sync')) {
        wp_schedule_event(time(), 'twicedaily', 'carmay_cron_trending_sync');
    }

    add_action('carmay_cron_trending_sync', 'carmay_process_trending_data');

    function carmay_process_trending_data() {
        // Garantizar que la etiqueta existe con nombre profesional
        $term = get_term_by('slug', 'trending', 'product_tag');
        if (!$term) {
            wp_insert_term('Tendencias 🔥', 'product_tag', ['slug' => 'trending']);
        }

        $categories = get_terms(['taxonomy' => 'product_cat', 'hide_empty' => true]);

        foreach ($categories as $cat) {
            $products = get_posts([
                'post_type' => 'product',
                'posts_per_page' => -1,
                'tax_query' => [['taxonomy' => 'product_cat', 'field' => 'term_id', 'terms' => $cat->term_id]],
                'meta_query' => [['key' => '_stock_status', 'value' => 'instock']]
            ]);

            if (empty($products)) continue;

            $data = [];
            foreach ($products as $p) {
                $data[] = [
                    'id' => $p->ID,
                    'v' => (int)get_post_meta($p->ID, 'total_sales', true),
                    's' => (int)get_post_meta($p->ID, '_stock', true)
                ];
                // Limpieza preventiva: quitamos la etiqueta antes del nuevo cálculo
                wp_remove_object_terms($p->ID, 'trending', 'product_tag');
            }

            // Ordenación y selección del 10%
            usort($data, function($a, $b) { return ($b['v'] <=> $a['v']) ?: ($b['s'] <=> $a['s']); });
            $trending_ids = array_column(array_slice($data, 0, max(1, ceil(count($data) * 0.10))), 'id');

            // Asignación final
            foreach ($trending_ids as $id) {
                wp_set_object_terms($id, 'trending', 'product_tag', true);
            }
        }
    }
})();

=== SNIPPET: detectar moneda para asignar a Pedido (ID: 2443) ===
(function() {
    // 1. CAPTURA HISTÓRICA: Se ejecuta una sola vez al crear el pedido
    add_action('woocommerce_checkout_create_order', function($order, $data) {
        global $WOOCS;
        
        // Intentamos obtener las tasas actuales del plugin FOX
        $currencies = isset($WOOCS) ? $WOOCS->get_currencies() : apply_filters('woocs_get_currencies', []);
        $current_currency = isset($WOOCS) ? $WOOCS->current_currency : 'CUP';

        if (isset($currencies['USD'])) {
            $raw_rate = $currencies['USD']['rate'];
            // Calculamos la tasa inversa (CUP por 1 USD) y redondeamos
            $tasa_fija = ($raw_rate > 0) ? round(1 / $raw_rate) : 0;

            // Guardamos los datos de forma permanente en el pedido
            $order->update_meta_data('_payment_currency_selected', sanitize_text_field($current_currency));
            $order->update_meta_data('_payment_rate_at_checkout', $tasa_fija);
        }
    }, 10, 2);

    // 2. VISTA DEL CLIENTE: Fila dentro de la tabla de totales (Gracias / Mi Cuenta / Emails)
    add_filter('woocommerce_get_order_item_totals', function($total_rows, $order, $tax_display) {
        $currency = $order->get_meta('_payment_currency_selected');
        $rate = $order->get_meta('_payment_rate_at_checkout');

        if ($currency && $rate) {
            $total_rows['payment_currency_row'] = array(
                'label' => 'Moneda a Pagar:',
                'value' => esc_html($currency) . ' (Cambio: 1 USD = ' . esc_html($rate) . ' CUP)'
            );
        }
        return $total_rows;
    }, 10, 3);

    // 3. VISTA DEL ADMINISTRADOR: Apartado en el detalle del pedido (Backend)
    add_action('woocommerce_admin_order_data_after_billing_address', function($order) {
        $currency = $order->get_meta('_payment_currency_selected');
        $rate = $order->get_meta('_payment_rate_at_checkout');

        if ($currency && $rate) {
            echo '<div class="address" style="margin-top: 20px; border-top: 1px solid #eee; padding-top: 10px;">';
            echo '<p><strong>DATOS DE CONVERSIÓN (HISTÓRICO)</strong></p>';
            echo '<p><strong>Moneda elegida:</strong> ' . esc_html($currency) . '</p>';
            echo '<p><strong>Tasa al momento del pedido:</strong> 1 USD = ' . esc_html($rate) . ' CUP</p>';
            echo '</div>';
        }
    }, 10, 1);
})();

(function() {
    // 1. Agregar la columna al panel de productos
    add_filter('manage_edit-product_columns', function($columns) {
        $new_columns = [];
        foreach ($columns as $key => $column) {
            $new_columns[$key] = $column;
            if ($key === 'price') {
                $new_columns['price_usd'] = 'Precio USD';
            }
        }
        return $new_columns;
    });

    // 2. Mostrar el valor convertido y la tasa redondeada
    add_action('manage_product_posts_custom_column', function($column, $post_id) {
        if ($column === 'price_usd') {
            global $WOOCS, $product;
            
            $current_product = is_a($product, 'WC_Product') ? $product : wc_get_product($post_id);
            
            if ($current_product) {
                $price_cup = $current_product->get_price();
                
                if ($price_cup) {
                    $currencies = $WOOCS->get_currencies();
                    
                    if (isset($currencies['USD'])) {
                        $rate = $currencies['USD']['rate'];
                        
                        // Calculamos el precio y redondeamos
                        $price_in_usd = round($price_cup * $rate);
                        
                        // Calculamos el valor de 1 USD en CUP para el mensaje de cambio y redondeamos
                        $tasa_inversa = round(1 / $rate);
                        
                        // Salida simple y uniforme con WooCommerce
                        echo wc_price($price_in_usd, array('currency' => 'USD', 'decimals' => 0));
                        echo '<br>Cambio: ' . $tasa_inversa;
                    } else {
                        echo '—';
                    }
                } else {
                    echo '—';
                }
            }
        }
    }, 10, 2);
})();

 * Valida que los campos de domicilio sean obligatorios si el checkbox está marcado.
 * La validación se ejecuta en el servidor antes de procesar el pago.
 */
function carmay_validar_domicilio_condicional() {
    
    // 1. Verificamos si el cliente marcó "Envío a Domicilio". 
    // Usamos 'billing_delivery_method' como la clave POST del checkbox.
    $quiere_domicilio = isset($_POST['billing_delivery_method']) && !empty($_POST['billing_delivery_method']);

    if ( $quiere_domicilio ) {
        
        // 2. Si quiere domicilio, revisamos manualmente que los campos CRÍTICOS no estén vacíos.
        
        // Validar Nombre del Receptor
        if ( empty($_POST['billing_receiver_name']) ) {
            wc_add_notice( 
                __( 'Si seleccionas envío a domicilio, debes escribir el <strong>Nombre del Receptor</strong>.', 'woocommerce' ), 
                'error' 
            );
        }

        // Validar Dirección (billing_address_1)
        if ( empty($_POST['billing_address_1']) ) {
            wc_add_notice( 
                __( 'Para el envío a domicilio, la <strong>Dirección</strong> es obligatoria.', 'woocommerce' ), 
                'error' 
            );
        }

        // Validar Teléfono del Receptor
        if ( empty($_POST['billing_receiver_phone']) ) {
            wc_add_notice( 
                __( 'Necesitamos un <strong>Teléfono de contacto</strong> para la entrega.', 'woocommerce' ), 
                'error' 
            );
        }
    }

// CARMAY OUTFIT - PROCESADOR DE URL (Global)
// Escucha parámetros en la URL y llena el carrito
// =========================================================================

add_action( 'template_redirect', function() {
    
    // 1. FILTRO RÁPIDO: Si no están nuestros parámetros, SALIR inmediatamente.
    // Esto ahorra recursos al servidor.
    if ( empty( $_GET['carmay_promo_ids'] ) ) {
        return;
    }

    // 2. DEPENDENCIAS: Si WooCommerce no está listo, abortar.
    if ( ! function_exists('WC') || ! WC()->cart ) {
        return;
    }

    // 3. OBTENER DATOS LIMPIOS
    $ids_raw = sanitize_text_field( $_GET['carmay_promo_ids'] );
    $coupon  = isset( $_GET['carmay_coupon'] ) ? sanitize_text_field( $_GET['carmay_coupon'] ) : '';
    $ids     = array_map( 'absint', explode( ',', $ids_raw ) );

    // 4. LIMPIEZA Y LLENADO
    WC()->cart->empty_cart(); // Vaciamos carrito anterior

    $added = false;
    foreach ( $ids as $id ) {
        if ( $id > 0 ) {
            // add_to_cart devuelve la llave del carrito si funciona, o false si falla
            if ( WC()->cart->add_to_cart( $id, 1 ) ) {
                $added = true;
            }
        }
    }

    // 5. APLICAR CUPÓN (Solo si se añadieron productos)
    if ( $added && ! empty( $coupon ) ) {
        if ( ! WC()->cart->has_discount( $coupon ) ) {
            WC()->cart->apply_coupon( $coupon );
        }
    }

    // 6. REDIRECCIÓN FINAL (Limpia la URL para que no se repita el proceso al recargar)
    // Redirigimos al checkout limpio
    wp_safe_redirect( wc_get_checkout_url() );
    exit;

}, 10 ); // Prioridad 10 es estándar y seguro para template_redirect
